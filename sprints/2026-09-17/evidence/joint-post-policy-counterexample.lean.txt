/- A read-only audit witness. No repository sources are modified.
   The authorization portal is the existing explicit inhabitation-only portal.
   The SOURCE predicate is additionally mandatory ModeEvidence on each family.
   Thus this does not purport to attack a deployed signature verifier. -/
namespace DreggJointPostPolicyCounterexample

open Minidregg.Theory
open Minidregg.Theory.CellState
open Minidregg.Theory.CellStateWitness
open Minidregg.Theory.TypedAuthorization
open Minidregg.Theory.TypedAuthorizationWitness

set_option autoImplicit false
noncomputable section

def view (state : LogicalState schemaB) : Minidregg.Pred.State :=
  ⟨[("left", if (state.fields false).isSome then 1 else 0),
    ("right", if (state.fields true).isSome then 1 else 0)]⟩

/-- The first-order source invariant: at least one of the two flags is zero. -/
def policy : Minidregg.Pred.Pred :=
  .any [.eq "left" 0, .eq "right" 0]

def patch (side : Bool) : Patch schemaB Digest where
  expectedPreRoot := cellB.root
  fieldFootprint := {side}
  resourceFootprint := ∅
  fieldWrites := [{ field := side, value := some () }]
  resourceWrites := []

def localPost (side : Bool) : LogicalState schemaB where
  fields := applyFieldWrites (patch side).fieldWrites cellB.logical.fields
  resources := cellB.logical.resources

def sourceRequest (side : Bool) : Request .object :=
  { requestB with nonce := if side then 29 else 28 }

def family (side : Bool) :
    SemanticEffectFamily.{0, 0, 0, 0, 0, 0} schemaB materializerB Unit where
  Declaration := Unit
  declarationCodec := AcceptedCellEffectWitness.unitCodec
  pre := cellB
  request := fun _ => ⟨.object, sourceRequest side⟩
  Outcome := fun _ => Unit
  outcomeCodec := fun _ => AcceptedCellEffectWitness.unitCodec
  ModeEvidence := fun _ _ => PLift
    (Minidregg.Pred.eval policy (view cellB.logical) (view (localPost side)) = true)
  effectDigest := fun _ => requestB.effectsDigest
  patch := fun _ _ => patch side
  nullifier := fun _ _ => none
  Release := fun _ _ => PEmpty
  DeclassificationAuthority := fun _ _ => PEmpty
  ReleaseAuthorization := fun _ _ release => release.elim
  DisclosureAllowed := fun _ _ decision => decision = .sealed

def authorization (side : Bool) :
    Authorized permissivePortal authState (sourceRequest side) where
  evidence := .proof () rfl
  policyWitness := ()
  policyMembershipWitness := ()
  policyEpochExact := rfl
  policyAddressExact := rfl
  policyMembershipVerified := rfl
  policyVerified := rfl

theorem localPolicy (side : Bool) :
    Minidregg.Pred.eval policy (view cellB.logical) (view (localPost side)) = true := by
  cases side <;> decide

theorem patchAccepted (side : Bool) :
    Nonempty (ValidatedPatch materializerB cellB (patch side)) := by
  have witness : ∃ validated : ValidatedPatch materializerB cellB (patch side),
      validate materializerB cellB (patch side) =
        ValidationOutcome.accepted validated := by
    unfold validate
    rw [dif_pos (show (patch side).expectedPreRoot = cellB.root from rfl)]
    rw [dif_pos (show (patch side).fieldFootprint = (patch side).namedFields by
      cases side <;> decide)]
    rw [dif_pos (show (patch side).resourceFootprint = (patch side).namedResources by
      simp [patch, Patch.namedResources])]
    exact ⟨_, rfl⟩
  exact ⟨witness.choose⟩

def accepted (side : Bool) :
    AcceptedCellEffect (portal := permissivePortal) (authState := authState)
      (family side) (sourceRequest side) cellB () () where
  authorization := authorization side
  preStateBound := rfl
  requestBound := rfl
  effectsDigestBound := rfl
  preRootBound := rfl
  modeEvidence := ⟨localPolicy side⟩
  validated := Classical.choice (patchAccepted side)
  disclosure := .sealed
  disclosureAllowed := rfl

def projection : AuthorizationProjection schemaB where
  project := fun _ => authState

def leg (side : Bool) : Leg permissivePortal authState cellB where
  Nullifier := Unit
  family := family side
  kind := .object
  request := sourceRequest side
  declaration := ()
  outcome := ()
  accepted := accepted side

def declaration :
    Declaration schemaB materializerB permissivePortal projection Bool where
  pre := cellB
  apex := ⟨0⟩
  legs := leg
  composition := { fieldMode := .disjoint, order := [false, true] }

def law : ResourceLaw schemaB materializerB permissivePortal Unit Int where
  stateDelta := fun _ _ _ _ _ => 0

theorem shapeValid : declaration.ShapeValid where
  orderComplete := ⟨by decide, fun i => by cases i <;> decide⟩
  resourcesDisjoint := by
    intro left right _
    simp [Declaration.legPatch, Leg.patch, declaration, leg, family, patch]
  fieldsValid := by
    intro left right different
    cases left <;> cases right
    · exact absurd rfl different
    · decide
    · decide
    · exact absurd rfl different

def commit : Commit law declaration where
  shape := shapeValid
  validated := declaration.jointValidated
  apexExact := rfl
  fieldsPreserved := by
    intro side field present
    change field ∈ ({side} : Finset Bool) at present
    have same := Finset.mem_singleton.mp present
    subst field
    cases side <;> rfl
  jointDeltaExact := by
    funext coordinate
    simp [Declaration.jointDelta, Declaration.aggregateDelta, ResourceLaw.delta, law]
  aggregateBalanced := by
    funext coordinate
    simp [Declaration.aggregateDelta, ResourceLaw.delta, law]

theorem sourcePoliciesHeldOnEveryAcceptedLocalPost (side : Bool) :
    Minidregg.Pred.eval policy (view cellB.logical)
      (view ((declaration.legs side).post.logical)) = true :=
  (accepted side).modeEvidence.down

theorem actualJointPostBreaksSameSourcePolicy :
    Minidregg.Pred.eval policy (view cellB.logical)
      (view commit.prepared.post.logical) = false := by decide

theorem disjointAcceptedCounterexample :
    Nonempty (Commit law declaration) ∧
      (∀ side, Minidregg.Pred.eval policy (view cellB.logical)
        (view ((declaration.legs side).post.logical)) = true) ∧
      Minidregg.Pred.eval policy (view cellB.logical)
        (view commit.prepared.post.logical) = false :=
  ⟨⟨commit⟩, sourcePoliciesHeldOnEveryAcceptedLocalPost,
    actualJointPostBreaksSameSourcePolicy⟩

#print axioms disjointAcceptedCounterexample

end
end DreggJointPostPolicyCounterexample
